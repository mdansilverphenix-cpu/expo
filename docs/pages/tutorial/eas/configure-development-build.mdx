import React from 'react';
import { NavigationContainer } from '@react-navigation/native';
import { createBottomTabNavigator } from '@react-navigation/bottom-tabs';
import { createStackNavigator } from '@react-navigation/stack';
import { StatusBar } from 'expo-status-bar';
import { Ionicons } from '@expo/vector-icons';

// Screens
import HomeScreen       from './screens/HomeScreen';
import RevenueScreen    from './screens/RevenueScreen';
import MethodsScreen    from './screens/MethodsScreen';
import PlannerScreen    from './screens/PlannerScreen';
import ProfileScreen    from './screens/ProfileScreen';
import AddRevenueScreen from './screens/AddRevenueScreen';
import MethodDetailScreen from './screens/MethodDetailScreen';
import GoalScreen       from './screens/GoalScreen';

// Theme
export const THEME = {
  dark:   '#1a1a2e',
  gold:   '#f0b030',
  gold2:  '#c89020',
  cream:  '#fff8f0',
  body:   '#2c2c2c',
  muted:  '#888888',
  white:  '#ffffff',
  green:  '#43a047',
  blue:   '#1e88e5',
  red:    '#e53935',
  purple: '#8e24aa',
  orange: '#fb8c00',
  card:   '#ffffff',
  bg:     '#f5f0e8',
};

const Tab = createBottomTabNavigator();
const Stack = createStackNavigator();

// ── Bottom Tab Navigator
function MainTabs() {
  return (
    <Tab.Navigator
      screenOptions={({ route }) => ({
        tabBarIcon: ({ focused, color, size }) => {
          const icons = {
            Accueil:  focused ? 'home'           : 'home-outline',
            Revenus:  focused ? 'wallet'         : 'wallet-outline',
            Méthodes: focused ? 'grid'           : 'grid-outline',
            Planning: focused ? 'calendar'       : 'calendar-outline',
            Profil:   focused ? 'person-circle'  : 'person-circle-outline',
          };
          return <Ionicons name={icons[route.name]} size={size} color={color} />;
        },
        tabBarActiveTintColor:   THEME.gold,
        tabBarInactiveTintColor: THEME.muted,
        tabBarStyle: {
          backgroundColor: THEME.dark,
          borderTopColor:  THEME.gold + '30',
          borderTopWidth:  1,
          paddingBottom:   8,
          paddingTop:      6,
          height:          62,
        },
        tabBarLabelStyle: {
          fontSize:    11,
          fontWeight:  '600',
          marginTop:   -2,
        },
        headerStyle:       { backgroundColor: THEME.dark },
        headerTintColor:   THEME.gold,
        headerTitleStyle:  { fontWeight: 'bold', fontSize: 18 },
      })}
    >
      <Tab.Screen name="Accueil"  component={HomeScreen}    options={{ title: '💰 Accueil' }} />
      <Tab.Screen name="Revenus"  component={RevenueScreen} options={{ title: '📊 Revenus' }} />
      <Tab.Screen name="Méthodes" component={MethodsScreen} options={{ title: '🎯 Méthodes' }} />
      <Tab.Screen name="Planning" component={PlannerScreen} options={{ title: '📅 Planning' }} />
      <Tab.Screen name="Profil"   component={ProfileScreen} options={{ title: '👤 Profil' }} />
    </Tab.Navigator>
  );
}

// ── Root Stack Navigator
export default function App() {
  return (
    <NavigationContainer>
      <StatusBar style="light" />
      <Stack.Navigator screenOptions={{ headerShown: false }}>
        <Stack.Screen name="Main"         component={MainTabs} />
        <Stack.Screen name="AddRevenue"   component={AddRevenueScreen} />
        <Stack.Screen name="MethodDetail" component={MethodDetailScreen} />
        <Stack.Screen name="Goal"         component={GoalScreen} />
      </Stack.Navigator>
    </NavigationContainer>
  );
}
